作业B：MemoryBot改造

```python
# ---- 三层记忆 Agent ------------------------------------------------------
class MemoryBot:
    def __init__(self, use_memory: bool = True):
        self.use_memory = use_memory
        self.summary = ""
        self.messages = []
        self.long_term = []
        self.round = 0

    def recall(self, query: str):
        """长期记忆 = 对话历史上的 RAG（不变）"""
        if not self.use_memory or not self.long_term:
            return []
        qv = embed_texts([query])[0]
        scored = sorted(
            ((cosine(qv, v), txt, r) for v, txt, r in self.long_term),
            key=lambda x: -x[0])
        return scored[:TOP_K]

    def _compress(self):
        """中期记忆（不变）"""
        old, keep = self.messages[:-MAX_RECENT], self.messages[-MAX_RECENT:]
        dialogue = "\n".join(f"{r}: {c}" for r, c in old)
        self.summary = summarize_llm(self.summary, dialogue)
        self.messages = keep

    def _write_memory(self, fact: str):
        """显式写入长期记忆：向量用 fact 原文（保证语义检索命中），展示文本带溯源"""
        self.long_term.append(
            (embed_texts([fact])[0], f"第{self.round}轮 用户要求记住: {fact}", self.round)
        )

    def chat(self, user_input: str):
        self.round += 1
        mems = self.recall(user_input)

        # 组装 prompt（同原来）
        parts = [SYSTEM]
        if mems:
            parts.append("【长期记忆·相关历史】\n" + "\n".join(f"- {txt}" for _, txt, _ in mems))
        if self.summary:
            parts.append("【对话摘要】\n" + self.summary)
        msgs = [{"role": "system", "content": "\n\n".join(parts)}]
        msgs += [{"role": r, "content": c} for r, c in self.messages]
        msgs.append({"role": "user", "content": user_input})

        # ★ 核心改动：默认不走 LLM 后不入库；只有显式"记住："才写长期记忆
        m = REMEMBER_RE.search(user_input)
        if m:
            fact = m.group(1).strip()
            self._write_memory(fact)
            reply = f"好的，我记住了：{fact}"     # 程序化确认，不依赖 LLM
        else:
            reply = chat_llm(msgs)

        # 短期记忆照常更新（user 原话仍进窗口，20轮后自然被截断/压缩）
        self.messages += [("user", user_input), ("assistant", reply)]
        if len(self.messages) > SUMMARY_TRIGGER:
            self._compress()

        mem_str = f"长期记忆{len(mems)}条" + (f"(top:{mems[0][0]:.2f})" if mems else "")
        state = f"[短期{len(self.messages)}条 | 摘要{'√' if self.summary else '×'} | 库内{len(self.long_term)}条 | {mem_str}]"
        return reply, state

```

通过添加正则匹配，让用户说"记住：xxx"时才把 xxx 写入长期记忆。

def format_docs(docs) -> str:
    return "\n\n".join(f"[片段{i+1}] {d.page_content}" for i, d in enumerate(docs))


# ---- 第 5 部分：演示入口 -------------------------------------------------
def answer(question: str) -> str:
    hits = vectorstore.similarity_search_with_score(question, k=3)  # 全程只检索这一次
    print("  [检索] 命中片段（相似度）:")
    for doc, score in hits:
        flag = "✓" if score >= THRESHOLD else f"✗ 低于{THRESHOLD}，过滤"
        print(f"    {score:.3f} {flag} {doc.page_content[:36]}…")

    valid = [d for d, s in hits if s >= THRESHOLD]        # ← 00 同款阈值过滤
    if not valid:
        top = hits[0][1] if hits else 0.0
        return (f"[阈值拦截] 最高分 {top:.3f} 未过 {THRESHOLD} → "
                f"检索层直接拒答，本次 LLM 调用数：0（省钱的防线）")

    print(f"  [放行] {len(valid)} 条过线 → 塞进 prompt 调 {LLM_MODEL}")
    return gen_chain.invoke({"context": format_docs(valid), "question": question})

思考题

1.为什么"摘要"保不住具体数字和名字？哪类信息必须进长期记忆而不是只进摘要？

摘要本身就是对信息进行高度概括，所以具体的数字和名字有时会丢失很正常。

用户明确要求记住的事情必须进记忆。还有就是比较重要的信息如：cookie、key、token还有账户密码等

2.对话记忆库和领域知识库要不要共用一个向量库？至少说两条理由。

不能用同一个向量库。
第一，领域知识库中存放的信息应该作为决策的关键信息，如果使用同一个，可能会污染领域知识库会影响决策准确性。

第二，如果我要删除历史对话内容只需要删除对话记忆库就行了，如果放一起就不好处理。

3.【成本】每轮全量重发全部历史，token 成本随轮数怎么涨？三层记忆各砍掉了哪部分的成本？

重发全部历史token成本肯定是随对话轮数越来越高的。

短期记忆：不变

中期记忆：减少了长上下文prompt的成本

长期记忆：减少了每轮都进行检索的成本
